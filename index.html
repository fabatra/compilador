4️⃣ Tomando decisões

numero = int(input("Digite um número: "))

if numero > 10:
    print("Maior que 10")
elif numero < 10:
    print("Menor que 10")
else:
    print("É igual a 10")

5️⃣ Número secreto

import random

numero_secreto = random.randint(1, 100)

print("Estou pensando em um número de 1 a 100.")

while True:
    tentativa = int(input("Digite seu palpite: "))

    if tentativa == numero_secreto:
        print("🎉 VOCÊ ACERTOU!")
        break
    elif tentativa < numero_secreto:
        print("⬆️ O número é MAIOR!")
    else:
        print("⬇️ O número é MENOR!")
