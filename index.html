Exercícios de Lógica de Programação em Python

1. Avaliação Condicional Simples

if numero > 10:
    print("Maior que 10")
elif numero < 10:
    print("Menor que 10")
else:
    print("É igual a 10")

2. Jogo do Número Secreto

import random

numero_secreto = random.randint(1, 100)
print("Estou pensando em um número de 1 a 100.")

while True:
    entrada = input("Digite seu palpite: ")
    
    # Tratamento para evitar que entradas vazias ou textos travem o programa
    try:
        tentativa = int(entrada)
    except ValueError:
        print("⚠️ Por favor, digite um número inteiro válido!")
        continue

    if tentativa == numero_secreto:
        print("🎉 VOCÊ ACERTOU!")
        break
    elif tentativa < numero_secreto:
        print("⬆️ O número é MAIOR!")
    else:
        print("⬇️ O número é MENOR!")
